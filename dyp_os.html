Cloud Sync Setup

Enter a Supabase project URL and anonymous key. Everyone using the same credentials and workspace ID will see the same live data.

Required Supabase table
create table dyp_workspace (
  workspace_id text primary key,
  payload jsonb not null,
  updated_at timestamptz default now()
);

alter table dyp_workspace enable row level security;
create policy "public read" on dyp_workspace for select using (true);
create policy "public insert" on dyp_workspace for insert with check (true);
create policy "public update" on dyp_workspace for update using (true);
